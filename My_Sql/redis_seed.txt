# --- 0. 环境变量占位（shell 层替换） ---
# ENV=dev
# RUNTIME_HOST=redis-runtime.internal
# CACHE_HOST=redis-cache.internal
# REDIS_PORT=6379
# REDIS_PASSWORD=***

# --- 1. 连接与版本检查（对两个实例分别执行） ---
PING
INFO server
INFO keyspace

# --- 2. 配置断言（只读，不改配置；期望值见注释） ---
CONFIG GET maxmemory
CONFIG GET maxmemory-policy
# runtime 实例期望：noeviction
# cache   实例期望：allkeys-lru
CONFIG GET appendonly
CONFIG GET appendfsync
CONFIG GET save

# --- 3. 示例种子（仅 dev/staging；与 SQL 种子 UUID 呼应） ---

# 3.1 K1 会话运行态镜像（runtime 实例）
HSET ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301 session_id 00000000-0000-0000-0000-000000000301 task_id "" user_id 00000000-0000-0000-0000-000000000302 mode free student_count 30 behavior_intensity mid classroom_theme blackboard bgm_level light planned_duration_s 600 status not_started consent_id 00000000-0000-0000-0000-000000000303 engine_tick_ms 1793491000000
EXPIRE ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301 6300

# 3.2 K2 会话序号（runtime 实例）
SET ai_teach:dev:classroom:session:seq:00000000-0000-0000-0000-000000000301 0
EXPIRE ai_teach:dev:classroom:session:seq:00000000-0000-0000-0000-000000000301 3600

# 3.3 K3 评测锁（runtime 实例；仅演示 token 写法，勿在生产预置）
SET ai_teach:dev:evaluation:lock:eval:00000000-0000-0000-0000-000000000301 demo-token-0001 NX EX 1800

# 3.4 K4 排行榜（cache 实例；2026-10 班级榜示例）
ZADD ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10 88.50 00000000-0000-0000-0000-000000000302
ZADD ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10 92.00 00000000-0000-0000-0000-000000000304
EXPIREAT ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10 1794096000
# 注：1794096000 = 2026-11-08 00:00:00 UTC（2026-10 周期末 + 7d 的示例值，生产按周期计算）

# 3.5 K5 题目缓存（cache 实例）
SET ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501 '{"topic_id":"00000000-0000-0000-0000-000000000501","subject_id":"00000000-0000-0000-0000-000000000201","stage":"junior_high","textbook_version":"部编版","topic_title":"背影","difficulty":3,"source":"platform","usage_count":0}' EX 3600

# --- 4. 从 MySQL 重建（模板；由对应服务执行，redis-cli 不直连 MySQL） ---
# 4.1 K1 session_state 重建
#   SQL: SELECT * FROM classroom.trial_session WHERE session_id = :sid AND deleted_at IS NULL;
#   Redis: HSET ai_teach:{env}:classroom:session:state:{sid} <字段...>; EXPIRE key 6300
# 4.2 K2 session_seq 重建
#   不可重建：Redis 丢失即从 1 重新计数（文档 §9）
# 4.3 K3 eval_lock 重建
#   无需重建：锁持有 <30min，丢失后下一轮重新获取
# 4.4 K4 leaderboard_rank 重建
#   SQL: SELECT user_id, score FROM growth_cert.leaderboard
#        WHERE scope=:scope AND scope_id=:sid AND period_key=:period ORDER BY rank_no;
#   Redis: DEL key; ZADD key score1 uid1 score2 uid2 ...; EXPIREAT key <period_end+7d>
# 4.5 K5 topic_cache 重建
#   SQL: SELECT * FROM task_content.trial_topic WHERE topic_id = :tid AND deleted_at IS NULL;
#   Redis: SET ai_teach:{env}:task:cache:topic:{tid} '<json>' EX 3600

# --- 5. 验证 ---
# 5.1 类型与 TTL
TYPE ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
TTL  ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
HGETALL ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
TYPE ai_teach:dev:classroom:session:seq:00000000-0000-0000-0000-000000000301
TTL  ai_teach:dev:classroom:session:seq:00000000-0000-0000-0000-000000000301
TYPE ai_teach:dev:evaluation:lock:eval:00000000-0000-0000-0000-000000000301
TTL  ai_teach:dev:evaluation:lock:eval:00000000-0000-0000-0000-000000000301
TYPE ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10
ZRANGE ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10 0 -1 WITHSCORES
TYPE ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501
TTL  ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501
GET  ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501

# 5.2 存在性
EXISTS ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
EXISTS ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10

# 5.3 扫描全部命名空间（禁止在生产用 KEYS，统一用 SCAN）
SCAN 0 MATCH ai_teach:dev:* COUNT 100

# 5.4 大 Key 抽检（Redis 8 支持 MEMORY USAGE）
MEMORY USAGE ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
MEMORY USAGE ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501
MEMORY USAGE ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10

# --- 6. 清理 / 回滚 ---
# 6.1 按 Key 精确清理（示例）
DEL ai_teach:dev:classroom:session:state:00000000-0000-0000-0000-000000000301
DEL ai_teach:dev:classroom:session:seq:00000000-0000-0000-0000-000000000301
DEL ai_teach:dev:evaluation:lock:eval:00000000-0000-0000-0000-000000000301
DEL ai_teach:dev:leaderboard:rank:class:00000000-0000-0000-0000-000000000401:2026-10
DEL ai_teach:dev:task:cache:topic:00000000-0000-0000-0000-000000000501
